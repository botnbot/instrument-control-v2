from django.db.models.signals import post_save
from django.dispatch import receiver
from .models import Instrument, Repair, Relocation

@receiver(post_save, sender=Repair)
def update_instrument_status_on_repair(sender, instance, created, **kwargs):
    instrument = instance.instrument
    if not instrument:
        return

    if created and not instance.date_of_receipt_from_repair:
        if instrument.status != Instrument.STATUS_IN_REPAIR:
            instrument.status = Instrument.STATUS_IN_REPAIR
            instrument.save()

    if instance.date_of_receipt_from_repair:
        if instrument.status != Instrument.STATUS_IN_USE:
            instrument.status = Instrument.STATUS_IN_USE
            instrument.save()

class RepairListView(ListView):
    model = Repair
    template_name = 'core/repairs/repairs_list.html'
    context_object_name = 'repairs'

    def get_queryset(self):
        queryset = super().get_queryset()

        # РђРЅРЅРѕС‚Р°С†РёСЏ: РїСЂРѕРґРѕР»Р¶РёС‚РµР»СЊРЅРѕСЃС‚СЊ РІ РґРЅСЏС… (РґР»СЏ Р·Р°РІРµСЂС€С‘РЅРЅС‹С… СЂРµРјРѕРЅС‚РѕРІ)
        queryset = queryset.annotate(
            duration_days=ExpressionWrapper(
                F('date_of_receipt_from_repair') - F('date_of_delivery_for_repair'),
                output_field=DurationField()
            )
        )

        sort = self.request.GET.get('sort', 'date_of_delivery_for_repair')
        order = self.request.GET.get('order', 'desc')

        if sort == 'duration':
            sort = 'duration_days'

        allowed_sort_fields = [
            'instrument__name',
            'repairer__name',
            'date_of_delivery_for_repair',
            'date_of_receipt_from_repair',
            'duration_days',
        ]
        if sort in allowed_sort_fields:
            if order == 'desc':
                sort = '-' + sort
            queryset = queryset.order_by(sort)

        return queryset

    def get_context_data(self, **kwargs):
        context = super().get_context_data(**kwargs)
        context['current_sort'] = self.request.GET.get('sort', 'date_of_delivery_for_repair')
        context['current_order'] = self.request.GET.get('order', 'desc')
        return context

@login_required
def return_from_repair(request, pk):
    instrument = get_object_or_404(Instrument, pk=pk)

    is_allowed = (
        request.user.is_superuser or
        request.user.groups.filter(name__in=['РђРґРјРёРЅРёСЃС‚СЂР°С†РёСЏ', 'РљР»Р°РґРѕРІС‰РёРєРё']).exists()
    )
    if not is_allowed:
        messages.error(request, "РЈ РІР°СЃ РЅРµС‚ РїСЂР°РІ РЅР° РІРѕР·РІСЂР°С‚ РёР· СЂРµРјРѕРЅС‚Р°.")
        return redirect('core:instrument_list')

    if instrument.status != Instrument.STATUS_IN_REPAIR:
        messages.error(request, "Р­С‚РѕС‚ РёРЅСЃС‚СЂСѓРјРµРЅС‚ РЅРµ РЅР°С…РѕРґРёС‚СЃСЏ РІ СЂРµРјРѕРЅС‚Рµ.")
        return redirect('core:instrument_list')

    if request.method == 'POST':
        date_of_receipt = request.POST.get('date_of_receipt_from_repair')
        repair_cost = request.POST.get('repair_cost')

        if not date_of_receipt:
            messages.error(request, "РЈРєР°Р¶РёС‚Рµ РґР°С‚Сѓ РІРѕР·РІСЂР°С‚Р° РёР· СЂРµРјРѕРЅС‚Р°.")
            return redirect('core:instrument_list')

        # РќР°С…РѕРґРёРј РїРѕСЃР»РµРґРЅРёР№ СЂРµРјРѕРЅС‚ СЌС‚РѕРіРѕ РёРЅСЃС‚СЂСѓРјРµРЅС‚Р°, Сѓ РєРѕС‚РѕСЂРѕРіРѕ РЅРµС‚ РґР°С‚С‹ РІРѕР·РІСЂР°С‚Р°
        repair = Repair.objects.filter(
            instrument=instrument,
            date_of_receipt_from_repair__isnull=True
        ).order_by('-date_of_delivery_for_repair').first()

        if repair:
            repair.date_of_receipt_from_repair = date_of_receipt
            if repair_cost:
                repair.repair_cost = repair_cost
            repair.save()
            messages.success(request, f'РРЅСЃС‚СЂСѓРјРµРЅС‚ "{instrument.name}" РІРѕР·РІСЂР°С‰С‘РЅ РёР· СЂРµРјРѕРЅС‚Р°.')
        else:
            messages.warning(request, "РќРµ РЅР°Р№РґРµРЅ Р°РєС‚РёРІРЅС‹Р№ СЂРµРјРѕРЅС‚ РґР»СЏ СЌС‚РѕРіРѕ РёРЅСЃС‚СЂСѓРјРµРЅС‚Р°.")

        instrument.status = Instrument.STATUS_IN_USE
        instrument.save()
        return redirect('core:instrument_list')
    return redirect('core:instrument_list')

    document.addEventListener('DOMContentLoaded', function() {

        const resultsBox = document.getElementById('search-results');
        const hiddenInput = document.getElementById('id_instrument');

        let timeout = null;

        searchInput.addEventListener('input', function() {
            clearTimeout(timeout);
            const query = this.value.trim();

            if (query.length < 2) {
                resultsBox.innerHTML = '';
                return;
            }

            // Р—Р°РґРµСЂР¶РєР° 300РјСЃ, С‡С‚РѕР±С‹ РЅРµ СЃР»Р°С‚СЊ Р·Р°РїСЂРѕСЃ РЅР° РєР°Р¶РґСѓСЋ Р±СѓРєРІСѓ

                fetch(`/api/search-instruments/?q=${encodeURIComponent(query)}`)
                    .then(response => response.json())
                    .then(data => {
                        resultsBox.innerHTML = '';
                        if (data.results.length === 0) {
                            resultsBox.innerHTML = '<div class="list-group-item text-muted">РќРёС‡РµРіРѕ РЅРµ РЅР°Р№РґРµРЅРѕ</div>';

                            btn.textContent = item.text;
                            btn.onclick = function() {
                                searchInput.value = item.text;
                                hiddenInput.value = item.id;
                                resultsBox.innerHTML = '';
                            };
                            resultsBox.appendChild(btn);
                        });
                    });
            }, 300);
        });

from django.contrib.auth.backends import ModelBackend
from django.contrib.auth import get_user_model

class RFIDBackend(ModelBackend):
    def authenticate(self, request, rfid_card=None, **kwargs):
        if rfid_card is None:
            return None
        UserModel = get_user_model()
        try:
            user = UserModel.objects.get(rfid_card=rfid_card)
        except UserModel.DoesNotExist:
            return None
        if user.is_active:
            return user
        return None
